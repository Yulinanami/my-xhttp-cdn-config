# XHTTP + CDN 搭建指南 (九)：卸载

本文档用于卸载本项目安装的 Xray、Nginx、ACME 和 Hysteria2。

> **注意**：以下指令会删除服务、配置、证书和订阅文件。移除的是默认安装路径，如果安装路径不是默认路径，需要自行调整。

## 1. Debian / Ubuntu

```bash
sudo bash <<'EOF'
systemctl stop xray nginx hysteria-server 2>/dev/null || true
systemctl disable xray nginx hysteria-server 2>/dev/null || true

rm -f /usr/local/bin/xray
rm -rf /usr/local/etc/xray
rm -rf /usr/local/share/xray
rm -rf /var/log/xray
rm -f /etc/systemd/system/xray.service

rm -f /usr/sbin/nginx
rm -rf /usr/local/nginx
rm -rf /etc/nginx
rm -rf /var/log/nginx
rm -f /etc/systemd/system/nginx.service

rm -f /usr/local/bin/hysteria
rm -rf /etc/hysteria
rm -f /etc/systemd/system/hysteria-server.service
rm -f /var/log/hysteria-server.log

crontab -l 2>/dev/null | grep -v ".acme.sh" | crontab - 2>/dev/null || true
rm -f /usr/local/bin/acme.sh
rm -rf /root/.acme.sh

rm -f /etc/ssl/private/private.key
rm -f /etc/ssl/private/fullchain.cer

rm -rf /etc/xhttp-cdn

for h in /root /home/*; do
  [ -d "$h" ] || continue
  rm -f "$h/client-config.txt"
  rm -f "$h/client-config-mihomo-full.yaml"
  rm -f "$h/client-config-mihomo-nodes.yaml"
  rm -f "$h/subscription-links.txt"
  rm -f "$h/subscription-v2rayn.png"
  rm -f "$h/subscription-mihomo-full.png"
  rm -f "$h/subscription-mihomo-nodes.png"
done

systemctl daemon-reload 2>/dev/null || true

echo "已移除 my-xhttp-cdn-config 的主要服务、配置、证书和订阅文件"
EOF
```
